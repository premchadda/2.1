Score Calculation Logic
Backend scoring algorithm
tests.js - Submit Handler 
// Score Calculation Algorithm
const totalQuestions = test.questions ?? questions.length;
const totalMarks = test.marks ?? (totalQuestions * 2);
const marksPerQuestion = totalQuestions > 0 ? totalMarks / totalQuestions : 0;
const negativeMarks = test.negativeMarking ?? 0;

questions.forEach((question) => {
  const selectedOption = answer?.selectedOption;
  const correctOption = question.correctAnswer;
  
  if (selectedOption === null) {
    unattempted++;
  } else if (selectedOption === correctOption) {
    correct++;
  } else {
    wrong++;
  }
});

// Final Score = (Correct × Marks) - (Wrong × Negative)
const score = (correct * marksPerQuestion) - (wrong * negativeMarks);
const accuracy = correct + wrong > 0 
  ? (correct / (correct + wrong)) * 100 
  : 0;

// Response Payload
{
  attemptId, testId, seriesId, testTitle,
  score: Math.max(0, score),
  totalMarks, totalQuestions,
  correct, wrong, unattempted,
  accuracy, timeSpent,
  rank, percentile, totalParticipants
}
