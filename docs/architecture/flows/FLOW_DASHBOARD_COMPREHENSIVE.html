User performance metrics
users.js - Analytics Handler
// Analytics calculation from results
const results = await dbHelpers.find('results', { 
  userId: req.user.id, 
  isCompleted: true 
});

// Aggregate totals
results.forEach((result) => {
  totalQuestions += result.totalQuestions || 0;
  totalCorrect += result.correctAnswers || 0;
  totalWrong += result.wrongAnswers || 0;
  totalScore += parseFloat(result.score) || 0;
  totalTimeSpent += result.timeSpent || 0;
});

// Calculate derived metrics
const avgAccuracy = totalQuestions > 0 
  ? Math.round((totalCorrect / totalQuestions) * 100) 
  : 0;

const avgScore = totalTests > 0 
  ? (totalScore / totalTests).toFixed(1) 
  : 0;

// Subject-wise breakdown
const subjectWise = Object.entries(subjectStats).map(([name, stats]) => ({
  name,
  accuracy: stats.total > 0 
    ? Math.round((stats.correct / stats.total) * 100) 
    : 0,
  attempted: stats.attempted
}));
