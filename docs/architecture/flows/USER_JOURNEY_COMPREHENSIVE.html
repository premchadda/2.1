🔄 ID System Architecture

Important: The system uses a hybrid ID approach for flexibility: 

// ID Types Used in the System:

// 1. Database Primary Keys (PostgreSQL SERIAL)
id: 1, 2, 3...  // Auto-incrementing integers

// 2. URL-Safe Slugs (for routing)
slug: "rrb-ntpc-2026-mega"  // Human-readable identifiers

// 3. Frontend ID Mapping
_id: 4  // Mapped from database id for frontend compatibility
id: "rrb-ntpc-2026-mega"  // Slug exposed as id to frontend

// 4. Enrollment Array Storage (PostgreSQL integer[])
enrolled_series: "{1,2,3}"  // PostgreSQL array format
// Parsed to: [1, 2, 3]
⚠️

Critical: ID Handling

Frontend should NEVER send raw integer IDs for operations. Always use slugs or stringified IDs. Backend converts to appropriate format for database queries.

router.post('/enroll/:seriesId', protect, async (req, res) => {
  // 1. Validate series exists
  const series = await dbHelpers.findById('testSeries', seriesId);
  
  // 2. Check current enrollment status
  const user = await dbHelpers.findById('users', req.user.id);
  const enrolledSeries = parseEnrolledSeries(user.enrolledSeries);
  
  // 3. Prevent duplicate enrollment
  if (enrolledSeries.includes(seriesId)) {
    return res.status(400).json({ message: 'Already enrolled' });
  }
  
  // 4. Add to enrolled_series array (PostgreSQL integer[])
  const newEnrolledSeries = [...enrolledSeries, seriesId];
  await dbHelpers.updateById('users', req.user.id, { 
    enrolledSeries: newEnrolledSeries 
  });
  
  res.json({ success: true, message: 'Enrolled successfully' });
});

// Test Submission Logic (tests.js)
router.put('/:testId/submit', protect, async (req, res) => {
  const { answers, timeSpent, attemptId } = req.body;
  
  // Calculate score
  let correct = 0, wrong = 0, unattempted = 0;
  
  questions.forEach((question) => {
    const userAnswer = answers.find(a => a.questionId === question.id);
    if (!userAnswer?.selectedOption) {
      unattempted++;
    } else if (userAnswer.selectedOption === question.correct) {
      correct++;
    } else {
      wrong++;
    }
  });
  
  // Calculate score with negative marking
  const score = (correct * marksPerQuestion) - (wrong * negativeMarks);
  const accuracy = (correct / (correct + wrong)) * 100;
  
  // Save attempt
  await dbHelpers.insertOne('attempts', {
    userId, testId, seriesId,
    score, correct, wrong, unattempted,
    accuracy, timeSpent, answers,
    isCompleted: true,
    submittedAt: new Date()
  });
});

🔒 Security Architecture

🔐

JWT Authentication

Tokens signed with JWT_SECRET, stored in httpOnly cookies. Token includes user ID for database lookup. 7-day expiry with refresh capability.

🛡️

Route Protection Middleware

• protect - Requires valid JWT token
• optionalAuth - Attaches user if token present
• admin - Requires admin role
• proPass - Requires active Pro subscription

✅

Input Validation

All user inputs validated with express-validator. SQL injection prevented through parameterized queries. XSS prevented through output encoding.

🔑

Password Security

Bcrypt hashing with salt rounds of 10. Passwords never returned in API responses. Reset tokens expire after 1 hour.

// Auth Middleware (auth/index.js)
export const protect = async (req, res, next) => {
  let token;
  
  // Check Authorization header
  if (req.headers.authorization?.startsWith('Bearer')) {
    token = req.headers.authorization.split(' ')[1];
  }
  
  // Also check httpOnly cookie
  if (!token && req.cookies?.token) {
    token = req.cookies.token;
  }
  
  if (!token) {
    return res.status(401).json({ message: 'Not authorized' });
  }
  
  const decoded = jwt.verify(token, process.env.JWT_SECRET);
  const user = await dbHelpers.findById('users', decoded.id);
  
  if (!user || user.isActive === false) {
    return res.status(401).json({ message: 'User not found or inactive' });
  }
  
  req.user = user;
  next();
};

// Frontend Error Handling (dataService.js)
apiClient.interceptors.response.use(
  (response) => response,
  (error) => {
    if (error.response) {
      const { status, data } = error.response;
      
      switch (status) {
        case 400:
          return Promise.reject(new ValidationError(data?.message));
        case 401:
          localStorage.removeItem('token');
          window.dispatchEvent(new Event('unauthorized'));
          return Promise.reject(new AuthenticationError(data?.message));
        case 403:
          return Promise.reject(new AuthenticationError('Access forbidden'));
        case 404:
          return Promise.reject(new NotFoundError(data?.message));
        default:
          return Promise.reject(new DataError(data?.message));
      }
    }
  }
);
